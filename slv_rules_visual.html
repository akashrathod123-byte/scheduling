Query needed to fill the "TBD" cells above

One pass over secure.order_source aggregated to CMF, joined to the CMF dimension for listcode, then binned by active-contact count. Same universe as the R1 table (ex-distributor for the totals column if we want them to line up).

WITH cmf_activity AS (
  SELECT
    o.ship_to_cmf                AS cmf_id,
    COUNT(DISTINCT o.contact_id) AS active_contacts,
    COUNT(*)                     AS orders
  FROM secure.order_source o
  WHERE o.entry_date  >= 20250312
    AND o.entry_date  <  20260312
    AND o.cancel_indicator = 'N'
    AND o.hk_stat_cd       = 'ACTIVE'
    AND o.contact_id  IS NOT NULL
    AND o.ship_to_cmf NOT IN (130387900, 12121200, 144173500)
  GROUP BY o.ship_to_cmf
)
SELECT
  c.listcode,
  COUNT(*)                                                            AS total_cmfs,
  SUM(CASE WHEN ca.active_contacts <  10               THEN 1 ELSE 0 END) AS cmfs_lt_10,
  SUM(CASE WHEN ca.active_contacts BETWEEN 10 AND 25   THEN 1 ELSE 0 END) AS cmfs_10_25,
  SUM(CASE WHEN ca.active_contacts >  25               THEN 1 ELSE 0 END) AS cmfs_gt_25,
  SUM(ca.orders)                                                      AS total_orders,
  SUM(CASE WHEN ca.active_contacts <  10               THEN ca.orders ELSE 0 END) AS orders_lt_10,
  SUM(CASE WHEN ca.active_contacts BETWEEN 10 AND 25   THEN ca.orders ELSE 0 END) AS orders_10_25,
  SUM(CASE WHEN ca.active_contacts >  25               THEN ca.orders ELSE 0 END) AS orders_gt_25
FROM cmf_activity ca
JOIN <cmf_dim> c ON c.cmf_id = ca.cmf_id
WHERE c.listcode IN ('05','06','07','08','09','10','97')  -- R3 listcodes only
GROUP BY c.listcode
ORDER BY total_cmfs DESC;
